# Rows of Darwin's libSystem that cite the macOS SDK header they are read from
# (MacOSX27.0.sdk usr/include), where `__dead2` is `__attribute__((__noreturn__))`
# (sys/cdefs.h). Every row carries `.include=` and sits under a comment naming its source,
# which `tests/cited_rows.rs` requires (D406).

# unistd.h, macOS SDK: `void _exit(int) __dead2;`.
_exit=func
func._exit.args=1
func._exit.arg.0=int,status
func._exit.noreturn=true
func._exit.ret=void
func._exit.include=unistd.h

# stdlib.h, macOS SDK: `void _Exit(int) __dead2;`.
_Exit=func
func._Exit.args=1
func._Exit.arg.0=int,status
func._Exit.noreturn=true
func._Exit.ret=void
func._Exit.include=stdlib.h

# _assert.h, macOS SDK: `assert` calls `__assert_rtn(__func__, __FILE__, __LINE__, #e)`, declared
# `void __assert_rtn(const char *, const char *, int, const char *) __dead2 __cold`.
__assert_rtn=func
func.__assert_rtn.args=4
func.__assert_rtn.arg.0=const char *,function
func.__assert_rtn.arg.1=const char *,file
func.__assert_rtn.arg.2=int,line
func.__assert_rtn.arg.3=const char *,assertion
func.__assert_rtn.noreturn=true
func.__assert_rtn.ret=void
func.__assert_rtn.include=_assert.h

# err.h, macOS SDK: `void errc(int, int, const char *, ...) __cold __dead2 __printflike(3, 4);`.
# No glibc, bionic or musl header declares it, so it is libSystem's alone.
errc=func
func.errc.args=4
func.errc.arg.0=int,eval
func.errc.arg.1=int,code
func.errc.arg.2=const char *,fmt
func.errc.arg.3=,...
func.errc.noreturn=true
func.errc.ret=void
func.errc.include=err.h
func.errc.tail=format(printf,2)
